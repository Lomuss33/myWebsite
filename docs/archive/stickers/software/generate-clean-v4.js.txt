const mix = (hex, target, amount) => '#' + [1, 3, 5].map(index =>
    Math.round(parseInt(hex.slice(index, index + 2), 16) * (1 - amount) + target * amount).toString(16).padStart(2, '0')).join('')

const shape = (d, fill, width = 10) =>

const rect = (x, y, w, h, radius, fill) => shape(`M${x + radius} ${y}H${x + w - radius}Q${x + w} ${y} ${x + w} ${y + radius}V${y + h - radius}Q${x + w} ${y + h} ${x + w - radius} ${y + h}H${x + radius}Q${x} ${y + h} ${x} ${y + h - radius}V${y + radius}Q${x} ${y} ${x + radius} ${y}Z`, fill)
const circle = (x, y, r, fill) => `<circle cx="${x}" cy="${y}" r="${r}" fill="${fill}"/>`
const disc = (x, y, r, fill) => `${circle(x, y, r + 8.5, 'url(#edge)')}${circle(x, y, r + 5, c.ink)}${circle(x, y, r, paint(fill))}`
const line = (d, stroke, width = 10) => `<path d="${d}" fill="none" stroke="${stroke}" stroke-width="${width}" stroke-linecap="round" stroke-linejoin="round"/>`
const rail = (d, stroke, width) => line(d, 'url(#edge)', width + 14) + line(d, c.ink, width + 8) + line(d, paint(stroke), width)
const glint = d => `<g opacity=".35">${line(d, c.ice, 3.5)}</g>`

// These PNGs are optional local artwork previews, never application assets.
if(process.argv.includes('--preview')) {
    const {default: sharp} = await import('sharp')
    const previewRoot = path.join(root, 'docs/tmp/software-stickers-v4')
    await fs.mkdir(previewRoot, {recursive: true})
    for(let project = 1; project <= 7; project++) {
        const pair = entries.filter(entry => entry.id === project)
        const layers = []
        for(let row = 0; row < 2; row++) {
            for(let col = 0; col < 2; col++) {
                const x = col * 480
                const y = row * 304
                const panel = `<svg width="480" height="304"><rect width="480" height="304" fill="${col ? '#18243A' : '#FFF5F7'}"/><text x="18" y="25" fill="${col ? '#D0DFEF' : '#17263F'}" font-family="sans-serif" font-size="13">${pair[row].title} · ${col ? 'dark' : 'light'}</text></svg>`
                layers.push({input: Buffer.from(panel), left: x, top: y})
                for(const [i, size] of [40, 104, 256].entries()) {
                    const image = await sharp(path.join(output, pair[row].filename)).resize(size, size).png().toBuffer()
                    // Match the CSS filter order on the rasterised size, including
                    // its fixed 2px shadow, rather than scaling a large shadow down.
                    const [brightness, saturation, contrast, shadow, opacity] = col ?
                        [1.12, 1.06, 1.02, '#091222', .32] : [.97, 1.08, 1.08, '#0F172A', .20]
                    const themed = `<svg xmlns="http://www.w3.org/2000/svg" width="${size + 12}" height="${size + 12}"><defs><filter id="theme" x="-50%" y="-50%" width="200%" height="200%" color-interpolation-filters="sRGB"><feComponentTransfer>${['R', 'G', 'B'].map(channel => `<feFunc${channel} type="linear" slope="${brightness}"/>`).join('')}</feComponentTransfer><feColorMatrix type="saturate" values="${saturation}"/><feComponentTransfer>${['R', 'G', 'B'].map(channel => `<feFunc${channel} type="linear" slope="${contrast}" intercept="${(1 - contrast) / 2}"/>`).join('')}</feComponentTransfer><feDropShadow dx="0" dy="2" stdDeviation="2" flood-color="${shadow}" flood-opacity="${opacity}"/></filter></defs><image x="6" y="6" width="${size}" height="${size}" href="data:image/png;base64,${image.toString('base64')}" filter="url(#theme)"/></svg>`
                    layers.push({input: Buffer.from(themed), left: x + [18, 90, 214][i] - 6, top: y + 34 + Math.round((256 - size) / 2)})
                }
            }
        }
        await sharp({create: {width: 960, height: 608, channels: 4, background: '#17263F'}}).composite(layers).png().toFile(path.join(previewRoot, `project-${project}.png`))
    }
}
